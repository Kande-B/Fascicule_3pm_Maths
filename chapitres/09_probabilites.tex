\section{Vocabulaire des Probabilités \& Événements}

\begin{cadredef}[Expérience Aléatoire \& Issues]
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
    \item Une \textbf{expérience aléatoire} est une expérience reproductible dont on connaît toutes les possibilités, mais dont le résultat dépend du hasard (impossible à prévoir à l'avance).
    \item Chaque résultat possible est appelé une \textbf{issue}.
    \item Un \textbf{événement} est constitué d'une ou plusieurs issues (ex : $A$ : \og Obtenir un nombre pair \fg{}).
\end{itemize}
\end{cadredef}

\begin{cadredef}[Probabilité d'un Événement]
La \textbf{probabilité} d'un événement $A$, notée $P(A)$, est un nombre compris entre \textbf{0} et \textbf{1} :
\[ \mathbf{0 \le P(A) \le 1} \qquad (\text{ou entre } 0\text{ \%} \text{ et } 100\text{ \%}) \]
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
    \item \textbf{Événement impossible} : sa probabilité vaut $\mathbf{0}$ (ex : obtenir un 7 avec un dé à 6 faces).
    \item \textbf{Événement certain} : sa probabilité vaut $\mathbf{1}$ (ex : obtenir un nombre inférieur ou égal à 6).
    \item \textbf{Événement contraire $\bar{A}$} : l'événement qui se réalise lorsque $A$ ne se réalise pas.
    \[ \mathbf{P(\bar{A}) = 1 - P(A)} \]
\end{itemize}
\end{cadredef}

\begin{cadreexemple}
Si la probabilité qu'un colis subisse un retard est $P(R) = 0{,}08$ ($8\text{ \%}$), alors la probabilité que le colis arrive dans les délais (événement contraire $\bar{R}$) est :
\[ P(\bar{R}) = 1 - P(R) = 1 - \pointille[1.5cm] = \pointille[1.5cm] \qquad (\text{soit } \pointille[1.5cm]\text{ \%}) \]
\end{cadreexemple}

\section{Loi d'Équiprobabilité \& Arbre des Possibles}

\begin{cadredef}[Loi d'Équiprobabilité]
Lorsque toutes les issues d'une expérience ont \textbf{exactement la même probabilité} de se produire (dés équilibrés, tirage au hasard sans triche), on est en situation d'\textbf{équiprobabilité} :
\[ \mathbf{P(A) = \frac{\text{Nombre d'issues favorables à } A}{\text{Nombre total d'issues possibles}}} \]
\end{cadredef}

\begin{cadremethode}[Construire et Utiliser un Arbre Pondéré à Deux Épreuves]
Pour représenter une expérience à deux étapes (ex : lancer d'une pièce puis tirage d'un jeton) :
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
    \item Sur chaque branche, on inscrit la \textbf{probabilité} de l'événement correspondant.
    \item La somme des probabilités issues d'un même n\oe ud est toujours égale à \textbf{1}.
    \item \textbf{Règle du produit} : La probabilité d'une issue au bout d'un chemin est égale au \textbf{produit} des probabilités le long de ce chemin.
\end{itemize}
\end{cadremethode}

\begin{cadreretenu}[L'Essentiel du Chapitre 9]
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
    \item Une probabilité est toujours comprise entre $\mathbf{0}$ et $\mathbf{1}$ (jamais négative, jamais supérieure à $1$).
    \item Équiprobabilité : $P(A) = \frac{\text{issues favorables}}{\text{issues totales}}$.
    \item Événement contraire : $P(\bar{A}) = 1 - P(A)$.
    \item Arbre des possibles : on multiplie les probabilités le long de chaque chemin.
\end{itemize}
\end{cadreretenu}
