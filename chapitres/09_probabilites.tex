\section{Vocabulaire des Probabilités \& Événements}

\begin{cadredef}[Expérience Aléatoire \& Issues]
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
    \item Une \textbf{expérience aléatoire} est une expérience dont on connaît tous les résultats possibles, mais dont le résultat dépend du hasard (impossible à prévoir avec certitude).
    \item Chaque résultat possible est appelé une \textbf{issue}.
    \item Un \textbf{événement} est un ensemble d'issues (ex : $A$ : « Obtenir un nombre pair »).
\end{itemize}
\end{cadredef}

\begin{cadredef}[Probabilité d'un Événement]
La \textbf{probabilité} d'un événement $A$, notée $P(A)$, est un nombre compris entre \textbf{0} et \textbf{1} :
\[ \mathbf{0 \le P(A) \le 1} \qquad (\text{ou entre } 0\text{ \%} \text{ et } 100\text{ \%}) \]
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
    \item \textbf{Événement impossible} : sa probabilité vaut $\mathbf{0}$ (ex : obtenir un 7 avec un dé à 6 faces).
    \item \textbf{Événement certain} : sa probabilité vaut $\mathbf{1}$ (ex : obtenir un nombre $\le 6$).
    \item \textbf{Événement contraire $\bar{A}$} : l'événement « non $A$ », sa probabilité est : $\mathbf{P(\bar{A}) = 1 - P(A)}$.
\end{itemize}
\end{cadredef}

\begin{cadreexemple}
Si la probabilité qu'un colis arrive en retard est $P(R) = 0{,}08$ ($8\text{ \%}$), alors la probabilité que le colis arrive à l'heure ($\bar{R}$) est :
\[ P(\bar{R}) = 1 - P(R) = 1 - \pointille[1.5cm] = \pointille[1.5cm] \qquad (\text{soit } \pointille[1.5cm]\text{ \%}) \]
\end{cadreexemple}

\section{Situation d'Équiprobabilité}

\begin{cadredef}[Loi d'Équiprobabilité]
Lorsque toutes les issues d'une expérience aléatoire ont \textbf{exactement la même chance} de se produire (dés équilibrés, pièces non truquées, tirage au sort au hasard), on est en situation d'\textbf{équiprobabilité}.
\[ \mathbf{P(A) = \frac{\text{Nombre d'issues favorables à } A}{\text{Nombre total d'issues possibles}}} \]
\end{cadredef}

\begin{cadremethode}[Méthode : Écrire une Probabilité]
On lance un dé classique à 6 faces non truqué. On note $A$ l'événement : « Obtenir un multiple de 3 » (issues : $3$ et $6$).
\begin{enumerate}[leftmargin=1.5em, itemsep=2pt]
    \item Nombre d'issues favorables : $2$ (le $3$ et le $6$).
    \item Nombre total d'issues : $6$ ($1, 2, 3, 4, 5, 6$).
    \item Probabilité : $P(A) = \frac{2}{6} = \mathbf{\frac{1}{3}} \approx \mathbf{0{,}333} \approx \mathbf{33{,}3\text{ \%}}$.
\end{enumerate}
\end{cadremethode}

\section{Expériences à Deux Épreuves \& Arbre des Choix}

\begin{cadremethode}[Construire et Utiliser un Arbre Pondéré]
Pour représenter une expérience à deux étapes successives (ex : lancer une pièce puis tirer une carte) :
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
    \item Chaque branche porte la \textbf{probabilité} de l'événement associé.
    \item La somme des probabilités issues d'un même n\oe ud vaut toujours \textbf{1}.
    \item \textbf{Règle du produit} : La probabilité de l'issue au bout d'un chemin est égale au \textbf{produit} des probabilités rencontrées le long de ce chemin.
\end{itemize}
\end{cadremethode}

\begin{cadreretenu}[L'Essentiel du Chapitre 9]
\begin{itemize}[leftmargin=1.5em, itemsep=2pt]
    \item Une probabilité est toujours comprise entre $\mathbf{0}$ et $\mathbf{1}$ (jamais négative, jamais $>1$).
    \item Équiprobabilité : $P(A) = \frac{\text{issues favorables}}{\text{issues totales}}$.
    \item Événement contraire : $P(\bar{A}) = 1 - P(A)$.
    \item Arbre des possibles : on multiplie les probabilités le long d'un chemin.
\end{itemize}
\end{cadreretenu}
